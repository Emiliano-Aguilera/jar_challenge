cmake_minimum_required(VERSION 3.8)
project(equipo_jar_navegacion)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
set(deps nav2_core nav2_costmap_2d nav2_util nav2_behaviors nav2_msgs rclcpp rclcpp_lifecycle pluginlib tf2_ros tf2_geometry_msgs geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs)
foreach(dep ${deps})
  find_package(${dep} REQUIRED)
endforeach()
add_library(jar_passage SHARED src/controller.cpp src/guards.cpp)
target_include_directories(jar_passage PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(jar_passage ${deps})
find_library(spin_library nav2_spin_behavior PATHS "${nav2_behaviors_DIR}/../../../lib" REQUIRED)
find_library(backup_library nav2_back_up_behavior PATHS "${nav2_behaviors_DIR}/../../../lib" REQUIRED)
target_link_libraries(jar_passage ${spin_library} ${backup_library})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS jar_passage ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
set(test_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../pruebas/pasaje_angosto")
if(BUILD_TESTING AND EXISTS "${test_dir}/test_geometry.cpp")
  enable_testing()
  foreach(test geometry passage)
    add_executable(test_${test} ${test_dir}/test_${test}.cpp)
    target_include_directories(test_${test} PRIVATE include)
    # These behavioral assertions must run in Release builds too.
    target_compile_options(test_${test} PRIVATE -UNDEBUG)
    add_test(NAME passage_${test} COMMAND test_${test})
  endforeach()
  add_executable(test_plugins ${test_dir}/test_plugins.cpp)
  target_include_directories(test_plugins PRIVATE include)
  target_compile_options(test_plugins PRIVATE -UNDEBUG)
  ament_target_dependencies(test_plugins ${deps})
  target_link_libraries(test_plugins jar_passage ${spin_library} ${backup_library})
  add_test(NAME passage_plugins COMMAND test_plugins)
  set_tests_properties(passage_plugins PROPERTIES TIMEOUT 30)
endif()
ament_export_include_directories(include)
ament_export_libraries(jar_passage)
ament_export_dependencies(${deps})
ament_package()
